cmake_minimum_required(VERSION 3.21)
project(LuaRuntime VERSION 5.5.1 LANGUAGES C)
include(GNUInstallDirs)
if(NOT EXISTS "${LUA_SOURCE_DIR}/lua.h")
  message(FATAL_ERROR "LUA_SOURCE_DIR must name the verified Lua 5.5.1 src directory")
endif()
file(GLOB sources CONFIGURE_DEPENDS "${LUA_SOURCE_DIR}/*.c")
list(REMOVE_ITEM sources "${LUA_SOURCE_DIR}/lua.c" "${LUA_SOURCE_DIR}/luac.c")
add_library(lua SHARED ${sources})
target_include_directories(lua PUBLIC "${LUA_SOURCE_DIR}")
target_compile_options(lua PRIVATE -std=c89 -Wall -Wextra -Wno-long-long)
# Keep upstream's normal 64-bit integer ABI on armhf too. LUA_USE_C89 selects
# 32-bit long there; GNU inline spelling supplies C89 syntax without that ABI change.
target_compile_definitions(lua PRIVATE _POSIX_C_SOURCE=200809L inline=__inline__)
if(APPLE)
  target_compile_definitions(lua PRIVATE LUA_USE_MACOSX)
  set_target_properties(lua PROPERTIES INSTALL_NAME_DIR "@rpath")
else()
  target_compile_definitions(lua PRIVATE LUA_USE_LINUX)
endif()
target_link_libraries(lua PRIVATE m ${CMAKE_DL_LIBS})
set_target_properties(lua PROPERTIES VERSION 5.5.1 SOVERSION 5.5)
install(TARGETS lua LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES "${LUA_SOURCE_DIR}/lua.h" "${LUA_SOURCE_DIR}/luaconf.h"
  "${LUA_SOURCE_DIR}/lauxlib.h" "${LUA_SOURCE_DIR}/lualib.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
