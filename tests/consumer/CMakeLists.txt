cmake_minimum_required(VERSION 3.21)
project(CborConsumer LANGUAGES C)
include(CTest)
find_package(C89Cbor CONFIG REQUIRED)
find_package(Threads REQUIRED)
include("${RUNTIME_HELPER}")
foreach(kind static shared)
  add_executable(consumer_${kind} main.c)
  target_compile_options(consumer_${kind} PRIVATE -std=c89 -Wall -Wextra -Werror -Wpedantic -Wno-long-long -UNDEBUG)
  target_compile_definitions(consumer_${kind} PRIVATE EXPECTED_VERSION="${EXPECTED_VERSION}")
  target_link_libraries(consumer_${kind} PRIVATE C89Cbor::${kind})
  c89_runtime(consumer_${kind})
  add_test(NAME consumer_${kind} COMMAND consumer_${kind})
endforeach()
add_executable(example_copy "${SDK_PREFIX}/share/doc/C89Cbor/examples/copy.c")
target_compile_options(example_copy PRIVATE -std=c89 -Wall -Wextra -Werror -Wpedantic -Wno-long-long)
target_link_libraries(example_copy PRIVATE C89Cbor::static)
c89_runtime(example_copy)
add_executable(example_peer "${SDK_PREFIX}/share/doc/C89Cbor/examples/cbor-peer.c")
target_compile_definitions(example_peer PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(example_peer PRIVATE -std=c89 -Wall -Wextra -Werror -Wpedantic -Wno-long-long)
target_link_libraries(example_peer PRIVATE C89Cbor::static)
c89_runtime(example_peer)
list(APPEND CMAKE_FIND_ROOT_PATH "${SDK_PREFIX}")
get_filename_component(lonejson_PREFIX "${lonejson_DIR}/../../.." ABSOLUTE)
list(APPEND CMAKE_FIND_ROOT_PATH "${lonejson_PREFIX}")
set(ENV{PKG_CONFIG_LIBDIR} "${SDK_PREFIX}/lib/pkgconfig:${lonejson_PREFIX}/lib/pkgconfig:${CpktCore_PREFIX}/lib/pkgconfig")
set(ENV{PKG_CONFIG_PATH} "")
find_package(PkgConfig REQUIRED)
pkg_check_modules(CBOR REQUIRED IMPORTED_TARGET c89cbor)
add_executable(consumer_pkgconfig main.c)
target_compile_options(consumer_pkgconfig PRIVATE -std=c89 -Wall -Wextra -Werror -Wpedantic -Wno-long-long -UNDEBUG)
target_compile_definitions(consumer_pkgconfig PRIVATE EXPECTED_VERSION="${EXPECTED_VERSION}")
target_link_libraries(consumer_pkgconfig PRIVATE PkgConfig::CBOR)
c89_runtime(consumer_pkgconfig)
add_test(NAME consumer_pkgconfig COMMAND consumer_pkgconfig)
foreach(kind static shared pkgconfig)
  add_executable(api_consumer_${kind} "${API_SURFACE_SOURCE}" "${API_SURFACE_STACK}")
  target_compile_options(api_consumer_${kind} PRIVATE -std=c89 -Wall -Wextra -Werror -Wpedantic -Wno-long-long -UNDEBUG)
  if(kind STREQUAL pkgconfig)
    target_link_libraries(api_consumer_${kind} PRIVATE PkgConfig::CBOR Threads::Threads)
  else()
    target_link_libraries(api_consumer_${kind} PRIVATE C89Cbor::${kind} Threads::Threads)
  endif()
  c89_runtime(api_consumer_${kind})
  add_test(NAME api_consumer_${kind} COMMAND api_consumer_${kind})
endforeach()
get_property(consumer_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${consumer_tests} PROPERTIES TIMEOUT 30 ENVIRONMENT "GLIBC_TUNABLES=glibc.pthread.stack_cache_size=0")
