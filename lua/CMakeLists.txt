cmake_minimum_required(VERSION 3.21)
project(C89CborLua LANGUAGES C)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(GNUInstallDirs)
include(CTest)
find_package(C89Cbor CONFIG REQUIRED)
get_filename_component(source_root "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
list(PREPEND CMAKE_MODULE_PATH "${source_root}/cmake")
find_package(Lua 5.5.1 EXACT MODULE REQUIRED)
include("${source_root}/cmake/Runtime.cmake")
add_library(cbor MODULE cbor.c)
set_target_properties(cbor PROPERTIES PREFIX "" C_VISIBILITY_PRESET hidden BUILD_RPATH_USE_ORIGIN ON)
if(NOT APPLE)
  target_link_options(cbor PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
  set_property(TARGET cbor APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
else()
  target_link_options(cbor PRIVATE "-Wl,-undefined,dynamic_lookup")
endif()
target_link_libraries(cbor PRIVATE C89Cbor::shared)
target_include_directories(cbor SYSTEM PRIVATE "${LUA_INCLUDE_DIR}")
target_compile_options(cbor PRIVATE -std=c89 -Wall -Wextra -Werror -Wpedantic -Wno-long-long -Wdeclaration-after-statement)
target_compile_definitions(cbor PRIVATE _POSIX_C_SOURCE=200809L)
install(TARGETS cbor LIBRARY DESTINATION lib/lua/5.5)
install(FILES cbor.lua DESTINATION share/lua/5.5/cbor-doc)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(cose_lua_benchmark ../bench/cose.c ../bench/measure.c)
  target_compile_definitions(cose_lua_benchmark PRIVATE C89_COSE_LUA _POSIX_C_SOURCE=200809L)
  target_compile_options(cose_lua_benchmark PRIVATE -std=c89 -Wall -Wextra -Werror -Wpedantic -Wno-long-long -UNDEBUG)
  target_include_directories(cose_lua_benchmark SYSTEM PRIVATE "${LUA_INCLUDE_DIR}")
  target_link_libraries(cose_lua_benchmark PRIVATE C89Cbor::shared ${LUA_LIBRARIES} ${CMAKE_DL_LIBS})
  c89_runtime(cose_lua_benchmark)
  add_test(NAME lua-cose-benchmark-smoke COMMAND cose_lua_benchmark gcm stream 16 "$<TARGET_FILE:cbor>")
  foreach(profile ed25519 p256 hmac)
    foreach(mode stream cwt)
      add_test(NAME lua-benchmark-${profile}-${mode} COMMAND cose_lua_benchmark ${profile} ${mode} 16 "$<TARGET_FILE:cbor>")
    endforeach()
  endforeach()
  add_executable(lua_test ../tests/lua_test.c ../tests/stack.c)
  target_link_libraries(lua_test PRIVATE C89Cbor::shared ${LUA_LIBRARIES} Threads::Threads ${CMAKE_DL_LIBS})
  target_include_directories(lua_test SYSTEM PRIVATE "${LUA_INCLUDE_DIR}")
  target_compile_options(lua_test PRIVATE -std=c89 -Wall -Wextra -Werror -Wpedantic -Wno-long-long -UNDEBUG)
  target_compile_definitions(lua_test PRIVATE _POSIX_C_SOURCE=200809L)
  c89_runtime(lua_test)
  add_test(NAME lua-api COMMAND lua_test "${CMAKE_CURRENT_BINARY_DIR}/cbor${CMAKE_SHARED_MODULE_SUFFIX}" "${source_root}/tests/lua_test.lua")
  foreach(group codec runtime policy)
    add_test(NAME lua-lifetime-${group} COMMAND lua_test "$<TARGET_FILE:cbor>" "${source_root}/tests/lua_lifetime_test.lua")
    set_tests_properties(lua-lifetime-${group} PROPERTIES TIMEOUT 30 LABELS "lua;integration;lifetime"
      ENVIRONMENT "GLIBC_TUNABLES=glibc.pthread.stack_cache_size=0;C89CBOR_LUA_LIFETIME_CASE=${group}")
  endforeach()
  add_test(NAME lua-exports COMMAND ${CMAKE_COMMAND} "-DNM=${CMAKE_NM}" "-DMODULE=$<TARGET_FILE:cbor>"
    "-DHEADER=${source_root}/include/c89cbor/lua.h" "-DDARWIN=${APPLE}"
    "-DINSPECT=$<IF:$<BOOL:${APPLE}>,${CMAKE_OTOOL},${CMAKE_READELF}>" -P "${source_root}/tests/lua_exports.cmake")
  set_tests_properties(lua-api PROPERTIES TIMEOUT 30 LABELS "lua;integration" ENVIRONMENT "GLIBC_TUNABLES=glibc.pthread.stack_cache_size=0")
endif()
