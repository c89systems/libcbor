function(c89_owned target)
  target_compile_options(${target} PRIVATE -std=c89 -Wall -Wextra -Werror -Wpedantic
    -Wno-long-long -Wstrict-prototypes -Wmissing-prototypes -Wshadow
    -Wpointer-arith -Wcast-qual -Wwrite-strings -Wdeclaration-after-statement)
  target_compile_definitions(${target} PRIVATE _POSIX_C_SOURCE=200809L)
endfunction()

  string(TOUPPER "${kind}" linkage)
  add_library(c89cbor_${kind} ${linkage} $<TARGET_OBJECTS:c89cbor_objects>)
  add_library(C89Cbor::${kind} ALIAS c89cbor_${kind})
  set_target_properties(c89cbor_${kind} PROPERTIES OUTPUT_NAME cbor EXPORT_NAME ${kind})
  target_include_directories(c89cbor_${kind} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
endforeach()
