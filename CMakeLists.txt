cmake_minimum_required(VERSION 3.21)
execute_process(COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/scripts/version.sh" "${C89CBOR_VERSION_OVERRIDE}"
  OUTPUT_VARIABLE project_version OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE version_rc)
if(NOT version_rc EQUAL 0)
  message(FATAL_ERROR "Unable to resolve project version")
endif()
project(C89Cbor VERSION "${project_version}" LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(C89CBOR_DISABLE_TESTS "Disable executable tests" OFF)
option(C89CBOR_BENCHMARKS "Build benchmarks" OFF)
find_package(lonejson 0.49.0 EXACT CONFIG REQUIRED)
function(c89_owned target)
  target_compile_options(${target} PRIVATE -std=c89 -Wall -Wextra -Werror -Wpedantic
    -Wno-long-long -Wstrict-prototypes -Wmissing-prototypes -Wshadow
    -Wpointer-arith -Wcast-qual -Wwrite-strings -Wdeclaration-after-statement)
  target_compile_definitions(${target} PRIVATE _POSIX_C_SOURCE=200809L)
endfunction()
add_library(c89cbor_objects OBJECT src/cbor.c src/io.c src/runtime.c)
c89_owned(c89cbor_objects)
set_target_properties(c89cbor_objects PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
target_include_directories(c89cbor_objects PRIVATE include)
target_link_libraries(c89cbor_objects PRIVATE lonejson::lonejson)
target_compile_definitions(c89cbor_objects PRIVATE C89CBOR_VERSION="${PROJECT_VERSION}")
foreach(kind static shared)
  string(TOUPPER "${kind}" linkage)
  add_library(c89cbor_${kind} ${linkage} $<TARGET_OBJECTS:c89cbor_objects>)
  add_library(C89Cbor::${kind} ALIAS c89cbor_${kind})
  set_target_properties(c89cbor_${kind} PROPERTIES OUTPUT_NAME cbor EXPORT_NAME ${kind})
  target_include_directories(c89cbor_${kind} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
endforeach()
target_link_libraries(c89cbor_static PUBLIC lonejson::lonejson_static)
target_link_libraries(c89cbor_shared PUBLIC lonejson::lonejson)
set_target_properties(c89cbor_shared PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
if(APPLE)
  set_target_properties(c89cbor_shared PROPERTIES INSTALL_NAME_DIR "@rpath")
else()
  target_link_options(c89cbor_shared PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/cmake/exports.map")
  set_property(TARGET c89cbor_shared APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/exports.map")
endif()
install(TARGETS c89cbor_static c89cbor_shared EXPORT C89CborTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE README.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DOCDIR}/docs)
install(FILES lua/cbor.lua DESTINATION ${CMAKE_INSTALL_DOCDIR}/lua)
install(FILES examples/copy.c DESTINATION ${CMAKE_INSTALL_DOCDIR}/examples)
install(EXPORT C89CborTargets NAMESPACE C89Cbor:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/C89Cbor)
configure_package_config_file(cmake/C89CborConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/C89CborConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/C89Cbor)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/C89CborConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/C89CborConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/C89CborConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/C89Cbor)
configure_file(cmake/c89cbor.pc.in c89cbor.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/c89cbor.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
if(BUILD_TESTING AND NOT C89CBOR_DISABLE_TESTS)
  find_package(Threads REQUIRED)
  foreach(kind static shared)
    add_executable(receiver_${kind}_test tests/receiver_test.c tests/stack.c)
    c89_owned(receiver_${kind}_test)
    target_compile_options(receiver_${kind}_test PRIVATE -UNDEBUG)
    target_link_libraries(receiver_${kind}_test PRIVATE c89cbor_${kind} Threads::Threads)
    add_test(NAME receiver-${kind} COMMAND receiver_${kind}_test)
  endforeach()
  add_executable(codec_test tests/codec_test.c)
  c89_owned(codec_test)
  target_compile_options(codec_test PRIVATE -UNDEBUG)
  target_link_libraries(codec_test PRIVATE c89cbor_static Threads::Threads)
  add_test(NAME codec COMMAND codec_test)
  set_tests_properties(codec PROPERTIES TIMEOUT 30 LABELS "unit;offline")
endif()

    add_executable(${name}_test tests/${name}_test.c tests/stack.c)
